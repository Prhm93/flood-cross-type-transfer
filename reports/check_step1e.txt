     1	"""
     2	exp3_controlled_synthetic.py
     3	============================
     4	
     5	THE CONFOUND-KILLER EXPERIMENT.
     6	
     7	The reviewer's strongest objection: your two real datasets differ in flood
     8	type AND solver AND region AND resolution AND terrain AND sample count.
     9	So a transfer failure cannot be attributed to flood type alone.
    10	
    11	This experiment removes every confound. We generate our own floods with:
    12	
    13	    SAME terrain generator      SAME grid size      SAME solver
    14	    SAME resolution             SAME time step      SAME sample count
    15	    SAME model architecture     SAME training budget
    16	
    17	and change EXACTLY ONE THING: how the water enters.
    18	
    19	    Mechanism A (point source): water enters at one cell, like a breach
    20	    Mechanism B (distributed):  water falls on every cell, like rainfall
    21	
    22	If transfer still fails here, the flood mechanism itself is the cause,
    23	because nothing else differs. That is the controlled result the paper needs.
    24	
    25	THE SOLVER
    26	----------
    27	We use the local inertial (or "simplified shallow water") formulation of
    28	Bates et al. (2010), which is the scheme used inside LISFLOOD-FP. It is a
    29	standard, citable, physically grounded method — not a toy.
    30	
    31	Flow between two neighbouring cells:
    32	
    33	    q_new = (q_old - g * h_flow * dt * dS) / (1 + g * dt * n^2 * |q| / h_flow^(7/3))
    34	
    35	where h_flow is the depth of water available to flow between the cells,
    36	dS is the water-surface slope, n is Manning's roughness, g is gravity.
    37	
    38	Then depth is updated by conservation of mass:
    39	
    40	    dh/dt = (inflow - outflow) / cell_area  + source
    41	
    42	The `source` term is where the two mechanisms differ, and it is the ONLY
    43	difference between them.
    44	
    45	Run:
    46	    python3 scripts/exp3_controlled_synthetic.py
    47	    python3 scripts/exp3_controlled_synthetic.py --quick   (smoke test size)
    48	"""

    72	def make_terrain(dim, seed, relief=2.0):
    73	    """
    74	    Generate gently undulating terrain by summing a few smooth waves.
    75	    Identical generator for both mechanisms — only the seed varies, and
    76	    we use THE SAME SEEDS for both, so terrain is matched pairwise.
    77	    """
    78	    rng = np.random.RandomState(seed)
    79	    y, x = np.mgrid[0:dim, 0:dim] / dim
    80	
    81	    z = np.zeros((dim, dim), dtype=np.float64)
    82	    for _ in range(4):
    83	        kx, ky = rng.uniform(1, 4, 2)
    84	        phase = rng.uniform(0, 2 * np.pi)
    85	        amp = rng.uniform(0.3, 1.0)
    86	        z += amp * np.sin(2 * np.pi * (kx * x + ky * y) + phase)
    87	
    88	    z -= z.min()
    89	    if z.max() > 0:
    90	        z = z / z.max() * relief
    91	    # gentle overall tilt so water has somewhere to go
    92	    z += 0.5 * x * relief * 0.3
    93	    return z.astype(np.float64)

    97	# SOLVER — local inertial shallow water (Bates et al. 2010)

   100	def simulate(z, source_field, n_steps, dt=2.0, dx=50.0, manning=0.05,
   101	             depth_floor=1e-4, save_every=None):
   102	    """
   103	    Run the flood simulation.
   104	
   105	    z            : (D,D) bed elevation, metres
   106	    source_field : (D,D) water added per second per cell, metres/second.
   107	                   This is the ONLY thing that differs between mechanisms.
   108	    n_steps      : number of solver steps
   109	    dt           : solver time step, seconds
   110	    dx           : cell size, metres
   111	    manning      : roughness
   112	
   113	    Returns depth (T,D,D), vx (T,D,D), vy (T,D,D) at the saved times.
   114	    """
   115	    D = z.shape[0]
   116	    h = np.zeros((D, D), dtype=np.float64)
   117	
   118	    # discharge per unit width on cell faces
   119	    qx = np.zeros((D, D + 1), dtype=np.float64)   # faces in x
   120	    qy = np.zeros((D + 1, D), dtype=np.float64)   # faces in y
   121	
   122	    if save_every is None:
   123	        save_every = max(1, n_steps // 40)
   124	
   125	    depths, vxs, vys = [], [], []
   126	
   127	    for step in range(n_steps):
   128	        wse = z + h   # water surface elevation
   129	
   130	        # ---- x-direction faces (between column i-1 and i) ----
   131	        dS = (wse[:, 1:] - wse[:, :-1]) / dx
   132	        hf = np.maximum(wse[:, 1:], wse[:, :-1]) - np.maximum(z[:, 1:], z[:, :-1])
   133	        hf = np.maximum(hf, 0.0)
   134	
   135	        q_old = qx[:, 1:-1]
   136	        num = q_old - G * hf * dt * dS
   137	        den = 1.0 + G * dt * manning ** 2 * np.abs(q_old) / \
   138	            np.maximum(hf, depth_floor) ** (7.0 / 3.0)
   139	        q_new = num / den
   140	        q_new = np.where(hf > depth_floor, q_new, 0.0)
   141	        qx[:, 1:-1] = q_new
   142	
   143	        # ---- y-direction faces ----
   144	        dS = (wse[1:, :] - wse[:-1, :]) / dx
   145	        hf = np.maximum(wse[1:, :], wse[:-1, :]) - np.maximum(z[1:, :], z[:-1, :])
   146	        hf = np.maximum(hf, 0.0)
   147	
   148	        q_old = qy[1:-1, :]
   149	        num = q_old - G * hf * dt * dS
   150	        den = 1.0 + G * dt * manning ** 2 * np.abs(q_old) / \
   151	            np.maximum(hf, depth_floor) ** (7.0 / 3.0)
   152	        q_new = num / den
   153	        q_new = np.where(hf > depth_floor, q_new, 0.0)
   154	        qy[1:-1, :] = q_new
   155	
   156	        # ---- mass conservation ----
   157	        # net flux into each cell = (flux in from left - flux out right) etc.
   158	        div = (qx[:, 1:] - qx[:, :-1]) / dx + (qy[1:, :] - qy[:-1, :]) / dx
   159	        h = h - dt * div + dt * source_field
   160	        h = np.maximum(h, 0.0)
   161	
   162	        if step % save_every == 0:
   163	            # cell-centred velocities from face discharges
   164	            qxc = 0.5 * (qx[:, :-1] + qx[:, 1:])
   165	            qyc = 0.5 * (qy[:-1, :] + qy[1:, :])
   166	            hsafe = np.maximum(h, depth_floor)
   167	            depths.append(h.copy())
   168	            vxs.append(qxc / hsafe)
   169	            vys.append(qyc / hsafe)
   170	
   171	    return (np.stack(depths).astype(np.float32),
   172	            np.stack(vxs).astype(np.float32),
   173	            np.stack(vys).astype(np.float32))

   177	# THE TWO MECHANISMS — the only difference

   179	
   180	def source_point(dim, rate_m3s, dx, seed):
   181	    """
   182	    MECHANISM A — point source (breach-like).
   183	    All the water enters through ONE cell.
   184	    """
   185	    rng = np.random.RandomState(seed + 10000)
   186	    field = np.zeros((dim, dim))
   187	    # place the source on the left edge, random row
   188	    r = rng.randint(dim // 4, 3 * dim // 4)
   189	    c = 1
   190	    field[r, c] = rate_m3s / (dx * dx)   # m3/s spread over one cell area
   191	    return field
   192	
   193	
   194	def source_distributed(dim, rate_m3s, dx, seed):
   195	    """
   196	    MECHANISM B — distributed source (rainfall-like).
   197	    The SAME total water volume, but spread over every cell.
   198	    """
   199	    field = np.full((dim, dim), rate_m3s / (dx * dx * dim * dim))
   200	    return field

   207	def generate_dataset(mechanism, n_sims, dim, n_steps, dt, dx, seed_offset=0,
   208	                     verbose=True):
   209	    """
   210	    Generate n_sims simulations for one mechanism, in the unified format.
   211	
   212	    Crucially, both mechanisms use THE SAME terrain seeds, so simulation i
   213	    of mechanism A has exactly the same terrain as simulation i of
   214	    mechanism B. Only the water source differs.
   215	    """
   216	    edge_index = _grid_edges(dim, dim)
   217	    samples = []
   218	
   219	    for i in range(n_sims):
   220	        terrain_seed = seed_offset + i        # SAME for both mechanisms
   221	        z = make_terrain(dim, terrain_seed)
   222	
   223	        rng = np.random.RandomState(terrain_seed + 500)
   224	        rate = rng.uniform(20.0, 60.0)         # m3/s, same range both ways
   225	
   226	        if mechanism == 'point':
   227	            src = source_point(dim, rate, dx, terrain_seed)
   228	        elif mechanism == 'distributed':
   229	            src = source_distributed(dim, rate, dx, terrain_seed)
   230	        else:
   231	            raise ValueError(mechanism)
   232	
   233	        depth, vx, vy = simulate(z, src, n_steps=n_steps, dt=dt, dx=dx)
   234	
   235	        T = depth.shape[0]
   236	        N = dim * dim
   237	
   238	        dyn = np.zeros((N, T, 4), dtype=np.float32)
   239	        dyn[:, :, 0] = depth.reshape(T, N).T
   240	        dyn[:, :, 1] = vx.reshape(T, N).T
   241	        dyn[:, :, 2] = vy.reshape(T, N).T
   242	        # channel 3 (the 'rainfall' slot) carries the source field for the
   243	        # distributed mechanism and zero for the point mechanism — exactly
   244	        # mirroring the real datasets' asymmetry
   245	        if mechanism == 'distributed':
   246	            dyn[:, :, 3] = float(src.mean())
   247	
   248	        zn = (z - z.min()) / max(z.max() - z.min(), 1e-9)
   249	
   250	        samples.append({
   251	            'nodes_static': zn.reshape(-1, 1).astype(np.float32),
   252	            'nodes_dynamic': dyn,
   253	            'edge_index': edge_index,
   254	            'num_nodes': N,
   255	            'num_steps': T,
   256	            'dataset': f'synth_{mechanism}',
   257	            'sample_id': terrain_seed,
   258	        })
   259	
   260	        if verbose and (i + 1) % 10 == 0:
   261	            wet = float((depth[-1] > 0.05).mean())
   262	            print(f"    {mechanism}: {i+1}/{n_sims} sims  "
   263	                  f"(last sim final wet fraction {wet:.3f}, "
   264	                  f"max depth {depth.max():.2f} m)")
   265	
   266	    return samples

   269	def normalise_together(list_of_sample_lists):
   270	    """Joint min-max normalisation across all provided sample lists."""
   271	    d_min = np.full(4, np.inf)
   272	    d_max = np.full(4, -np.inf)
   273	    for lst in list_of_sample_lists:
   274	        for s in lst:
   275	            d = s['nodes_dynamic']
   276	            for c in range(4):
   277	                d_min[c] = min(d_min[c], float(d[:, :, c].min()))
   278	                d_max[c] = max(d_max[c], float(d[:, :, c].max()))
   279	    rng_ = d_max - d_min
   280	    rng_[rng_ == 0] = 1.0
   281	
   282	    out = []
   283	    for lst in list_of_sample_lists:
   284	        new = []
   285	        for s in lst:
   286	            s2 = dict(s)
   287	            s2['nodes_dynamic'] = ((s['nodes_dynamic'] - d_min) / rng_).astype(np.float32)
   288	            new.append(s2)
   289	        out.append(new)
   290	    return out

   304	    ap.add_argument('--dt', type=float, default=2.0)
   305	    ap.add_argument('--dx', type=float, default=50.0)
   306	    ap.add_argument('--epochs', type=int, default=60)
   307	    ap.add_argument('--seeds', type=int, nargs='+', default=[0, 1, 2])
   308	    ap.add_argument('--out-dir', default='results')
   309	    args = ap.parse_args()
   310	
   311	    if args.quick:
   312	        args.dim, args.n_train, args.n_test = 24, 4, 3
   313	        args.solver_steps, args.epochs, args.seeds = 200, 3, [0]
   314	
   315	    os.makedirs(args.out_dir, exist_ok=True)
   316	    device = get_device()
   317	
   318	    print("=" * 66)
   319	    print("  EXPERIMENT 3 — CONTROLLED CROSS-MECHANISM TRANSFER")
   320	    print("  Same terrain, same solver, same grid, same everything.")
   321	    print("  ONLY the water source differs.")
   322	    print("=" * 66)
   323	    print(f"  grid {args.dim}x{args.dim}, {args.n_train} train + {args.n_test} "
   324	          f"test sims per mechanism, device {device}")
   325	
   326	    # ---- generate ----
   327	    t0 = time.time()
   328	    print("\n  Generating POINT-SOURCE floods (breach-like)...")
   329	    pt_train = generate_dataset('point', args.n_train, args.dim,
   330	                                 args.solver_steps, args.dt, args.dx, 0)
   331	    pt_test = generate_dataset('point', args.n_test, args.dim,
   332	                                args.solver_steps, args.dt, args.dx, 900)
   333	
   334	    print("\n  Generating DISTRIBUTED floods (rainfall-like), SAME terrain seeds...")
   335	    di_train = generate_dataset('distributed', args.n_train, args.dim,
   336	                                 args.solver_steps, args.dt, args.dx, 0)
   337	    di_test = generate_dataset('distributed', args.n_test, args.dim,
   338	                                args.solver_steps, args.dt, args.dx, 900)
   339	
   340	    print(f"\n  Generation took {time.time() - t0:.0f}s")
   341	
   342	    pt_train, pt_test, di_train, di_test = normalise_together(
   343	        [pt_train, pt_test, di_train, di_test])
   344	
   345	    data = {'point': {'train': pt_train, 'test': pt_test},
   346	            'distributed': {'train': di_train, 'test': di_test}}
   347	
   348	    # ---- train and cross-test ----
   349	    results = []
   350	    for train_on in ['point', 'distributed']:
   351	        away = 'distributed' if train_on == 'point' else 'point'
   352	        for seed in args.seeds:
   353	            print(f"\n  --- train on {train_on}, seed {seed} ---")
   354	            set_seed(seed)
   355	            model = build_model('vector', 64, 3, device)
   356	            ckpt = os.path.join(args.out_dir, f'synth_{train_on}_s{seed}.pt')
   357	            train_model(model, data[train_on]['train'], device,
   358	                        epochs=args.epochs, max_train_samples=args.n_train,
   359	                        ckpt_path=ckpt, log_every=max(1, args.epochs // 4))
   360	
   361	            home = evaluate(model, data[train_on]['test'], device,
   362	                            max_steps=30, max_samples=args.n_test)
   363	            awayr = evaluate(model, data[away]['test'], device,
   364	                             max_steps=30, max_samples=args.n_test)
   365	
   366	            retained = (100.0 * awayr['csi_final'] / home['csi_final']
   367	                        if home['csi_final'] > 0 else float('nan'))
   368	
   369	            print(f"    HOME ({train_on:11s}) CSI {home['csi_final']:.4f}  "
   370	                  f"vol ratio {home['volume_ratio']:.2f}")
   371	            print(f"    AWAY ({away:11s}) CSI {awayr['csi_final']:.4f}  "
   372	                  f"vol ratio {awayr['volume_ratio']:.2f}")
   373	            print(f"    retained {retained:.1f}%")
   374	
   375	            results.append({
   376	                'train_on': train_on, 'away': away, 'seed': seed,
   377	                'home': home, 'away_scores': awayr, 'retained_pct': retained,
   378	            })
   379	
   380	    # ---- summary ----
   381	    print("\n" + "=" * 66)
   382	    print("  CONTROLLED EXPERIMENT SUMMARY")
   383	    print("=" * 66)
   384	    for tr in ['point', 'distributed']:
   385	        rs = [r for r in results if r['train_on'] == tr]
   386	        if not rs:
   387	            continue
   388	        hm = np.mean([r['home']['csi_final'] for r in rs])
   389	        aw = np.mean([r['away_scores']['csi_final'] for r in rs])
   390	        rt = np.nanmean([r['retained_pct'] for r in rs])
   391	        print(f"  trained on {tr:12s}: home CSI {hm:.4f}  away CSI {aw:.4f}  "
   392	              f"retained {rt:.1f}%")
   393	
   394	    print("\n  INTERPRETATION: terrain, grid, solver, resolution and sample")
   395	    print("  count are IDENTICAL across the two mechanisms. Any transfer gap")
   396	    print("  here is attributable to the flood-driving mechanism alone.")
   397	
   398	    with open(os.path.join(args.out_dir, 'result_exp3_controlled.json'), 'w') as f:
   399	        json.dump({'config': vars(args), 'results': results}, f,
   400	                  indent=2, default=str)
   401	    print(f"\n  Saved {args.out_dir}/result_exp3_controlled.json")
